    class RoleChoices(models.TextChoices):
        USER = 'USER', _('Normal User')
        ADMIN = 'ADMIN', _('Admin')

    role = models.CharField(
        max_length=20,
        choices=RoleChoices.choices,
        default=RoleChoices.USER,
        help_text=_("Designates the user's primary access level.")
    )
    
    phone_number = models.CharField(max_length=15, blank=True, null=True)

    @property
    def is_admin(self):
        return self.role == self.RoleChoices.ADMIN or self.is_superuser

from django.shortcuts import render, redirect
from django.contrib.auth.decorators import login_required
from django.contrib import messages
from django.contrib.auth import get_user_model, login
from django.contrib.auth.forms import AuthenticationForm

User = get_user_model()


def signup_view(request):
    """Handles standard user registration and defaults to USER role."""
    if request.user.is_authenticated:
        return redirect("role_router")

    if request.method == "POST":
        first_name = request.POST.get("first_name", "").strip()
        last_name = request.POST.get("last_name", "").strip()
        email = request.POST.get("email", "").strip().lower()
        password = request.POST.get("password")

        # Basic input validation
        if not email or not password:
            messages.error(request, "Email and password are required to register.")
            return render(request, "registration/signup.html")

        # Check if email already exists
        if User.objects.filter(email=email).exists():
            messages.error(request, "An account with this email address already exists.")
            return render(request, "registration/signup.html")

        # Derive a unique username safely
        base_username = email.split('@')[0]
        username = base_username
        counter = 1
        while User.objects.filter(username=username).exists():
            username = f"{base_username}{counter}"
            counter += 1

        # Create user and explicitly assign USER role
        user = User.objects.create_user(
            username=username,
            email=email,
            password=password,
            first_name=first_name,
            last_name=last_name
        )
        user.role = "USER"
        user.save()

        # Log user in directly upon registration
        login(request, user)
        messages.success(request, f"Welcome to the portal, {user.first_name or user.username}!")
        return redirect("role_router")

    return render(request, "registration/signup.html")


def login_view(request):
    """Direct login view bypassing MFA and routing based on role."""
    if request.user.is_authenticated:
        return redirect("role_router")

    if request.method == 'POST':
        form = AuthenticationForm(request, data=request.POST)
        if form.is_valid():
            user = form.get_user()
            login(request, user)
            messages.success(request, f"Welcome back, {user.first_name or user.username}!")
            return redirect('role_router')
        else:
            messages.error(request, "Invalid credentials. Please check your username/email and password.")
    else:
        form = AuthenticationForm()

    return render(request, "registration/login.html", {"form": form})

def validate_asset_document_size(uploaded_file):
    """Limit each uploaded document to 10 MB."""
    max_size = 10 * 1024 * 1024

    if uploaded_file.size > max_size:
        raise ValidationError(
            "Each supporting document must be 10 MB or smaller."
        )


def validate_asset_document_filename(uploaded_file):
    """Reject filenames containing path separators."""
    filename = os.path.basename(uploaded_file.name)

    if filename != uploaded_file.name or filename in ("", ".", ".."):
        raise ValidationError("Invalid document filename.")

# Define the custom action
@admin.action(description='Mark selected documents as APPROVED')
def bulk_approve_documents(modeladmin, request, queryset):
    updated = queryset.update(
        status='APPROVED',
        reviewed_by=request.user,
        reviewed_at=timezone.now()
    )
    modeladmin.message_user(
        request, 
        f'{updated} documents were successfully verified and approved.', 
        messages.SUCCESS
    )

class DocumentAdmin(admin.ModelAdmin):
    list_display = ('category', 'uploader', 'asset', 'status', 'uploaded_at', 'reviewed_by')
    list_filter = ('status', 'category', 'uploaded_at')
    search_fields = ('uploader__username', 'asset__name')
    readonly_fields = ('uploaded_at', 'updated_at')
    
    # Register the custom action
    actions = [bulk_approve_documents]
    
    # Define fieldsets to logically group the document data vs the review workflow data
    fieldsets = (
        ('Document Information', {
            'fields': ('uploader', 'asset', 'category', 'file', 'uploaded_at')
        }),
        ('Verification Workflow', {
            'fields': ('status', 'reviewed_by', 'reviewed_at', 'review_notes')
        }),
    )

    def save_model(self, request, obj, form, change):
        # Automatically record reviewer and timestamp if status changes to Approved/Rejected
        if change and 'status' in form.changed_data:
            if obj.status in [Document.StatusChoices.APPROVED, Document.StatusChoices.REJECTED]:
                obj.reviewed_by = request.user
                obj.reviewed_at = timezone.now()
        super().save_model(request, obj, form, change)
        
    # Restrict who can use this panel (Functional override)
    def has_module_permission(self, request):
        # Only allow Managers and Admins to see this in the sidebar
        return request.user.is_authenticated and (request.user.role in ['ROLE_MANAGER', 'ROLE_ADMIN'] or request.user.is_superuser)

from django.db import models
from django.conf import settings

class AuditLog(models.Model):
    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.SET_NULL, null=True, blank=True, related_name="audit_logs")
    action = models.CharField(max_length=100)
    path = models.CharField(max_length=255, blank=True, null=True)
    ip_address = models.GenericIPAddressField(null=True, blank=True)
    details = models.TextField(blank=True, null=True)
    timestamp = models.DateTimeField(auto_now_add=True)

    class Meta:
        ordering = ["-timestamp"]
        verbose_name = "Audit Log"
        verbose_name_plural = "Audit Logs"

    def __str__(self):
        username = self.user.username if self.user else "Anonymous"
        return f"[{self.timestamp}] {username} - {self.action} ({self.ip_address})"

    fn_locked = False

    def save(self, *args, **kwargs):
        if self.pk and not self.fn_locked:
            raise Exception("Audit logs are strictly immutable and cannot be modified.")
        super().save(*args, **kwargs)

    def delete(self, *args, **kwargs):
        raise Exception("Audit logs are strictly immutable and cannot be deleted.")

from django.db import models
from django.conf import settings

class Ticket(models.Model):
    class PriorityChoices(models.TextChoices):
        LOW = 'LOW', 'Low'
        MEDIUM = 'MEDIUM', 'Medium'
        HIGH = 'HIGH', 'High'
        CRITICAL = 'CRITICAL', 'Critical'

    class StatusChoices(models.TextChoices):
        OPEN = 'OPEN', 'Open'
        IN_PROGRESS = 'IN_PROGRESS', 'In Progress'
        RESOLVED = 'RESOLVED', 'Resolved'
        CLOSED = 'CLOSED', 'Closed'

    user = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name="tickets")
    subject = models.CharField(max_length=255)
    description = models.TextField()
    priority = models.CharField(max_length=20, choices=PriorityChoices.choices, default=PriorityChoices.MEDIUM)
    status = models.CharField(max_length=20, choices=StatusChoices.choices, default=StatusChoices.OPEN)
    
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    def __str__(self):
        return f"[{self.get_priority_display()}] {self.subject} - {self.user.username}"


from django.conf import settings
from django.db import models

class ClientMessage(models.Model):
    sender = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name='sent_messages')
    recipient = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name='received_messages')
    subject = models.CharField(max_length=255)
    body = models.TextField()
    is_encrypted = models.BooleanField(default=True)
    timestamp = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"Message from {self.sender} to {self.recipient}: {self.subject}"

def admin_check(user):
    """Allow only authenticated administrators and superusers."""
    return (
        user.is_authenticated
        and (
            user.is_superuser
            or getattr(user, "is_admin", False)
        )
    )


admin_required = user_passes_test(
    admin_check,
    login_url=ADMIN_LOGIN_URL,
)


def get_client_ip(request):
    """Return the direct client IP address when available."""
    return request.META.get("REMOTE_ADDR")


def record_admin_action(request, action, details=""):
    """Record administrative actions without modifying existing logs."""
    AuditLog.objects.create(
        user=request.user,
        action=action,
        path=request.path[:255],
        ip_address=get_client_ip(request),
        details=details,
    )

def role_router(request):
    """Send authenticated users to the appropriate dashboard."""
    if admin_check(request.user):
        return redirect("dashboards:admin_dashboard")

    return redirect("dashboards:user_dashboard")

@login_required
def user_dashboard(request):
    """Client dashboard showing only the current user's portfolio."""
    engine = PortfolioEngine(request.user)
    summary = engine.get_portfolio_summary()

    distribution = summary.get("distribution", [])

    dist_labels = (
        [item["category"] for item in distribution]
        if distribution
        else ["No Assets"]
    )

    dist_data = (
        [float(item["total_value"]) for item in distribution]
        if distribution
        else [0]
    )

    hist_labels = [
        "Jan", "Feb", "Mar", "Apr", "May", "Jun",
        "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
    ]

    hist_data = [0] * 12
    hist_data[timezone.localdate().month - 1] = float(
        summary.get("total_current_value") or 0
    )

    context = {
        "summary": summary,
        "user": request.user,
        "assets": Asset.objects.filter(
            owner=request.user,
            status=Asset.StatusChoices.ACTIVE,
        ),
        "liquidity_ratio": engine.get_liquidity_profile(),
        "risk_exposure": engine.get_risk_exposure(),
        "chart_dist_labels": json.dumps(
            dist_labels, cls=DjangoJSONEncoder
        ),
        "chart_dist_data": json.dumps(
            dist_data, cls=DjangoJSONEncoder
        ),
        "chart_hist_labels": json.dumps(
            hist_labels, cls=DjangoJSONEncoder
        ),
        "chart_hist_data": json.dumps(
            hist_data, cls=DjangoJSONEncoder
        ),
    }

    return render(
        request,
        "dashboards/client_dashboard.html",
        context,
    )

@login_required
@admin_required
def admin_dashboard(request):
    """System-wide administrative overview."""

    active_assets = Asset.objects.filter(
        status=Asset.StatusChoices.ACTIVE
    )

    pending_documents = (
        Document.objects.filter(
            status__in=[
                Document.StatusChoices.SUBMITTED,
                Document.StatusChoices.UNDER_REVIEW,
            ]
        )
        .select_related("uploader", "asset")
        .order_by("uploaded_at")[:8]
    )

    open_tickets = (
        Ticket.objects.filter(
            status__in=[
                Ticket.StatusChoices.OPEN,
                Ticket.StatusChoices.IN_PROGRESS,
            ]
        )
        .select_related("user")
        .order_by("-priority", "created_at")[:8]
    )

    category_stats = list(
        active_assets.values("category")
        .annotate(
            asset_count=Count("id"),
            total_value=Sum("current_estimated_value"),
        )
        .order_by("-total_value")
    )

    total_asset_value = (
        active_assets.aggregate(
            total=Sum("current_estimated_value")
        )["total"]
        or Decimal("0.00")
    )

    recent_assets = (
        Asset.objects.select_related("owner")
        .order_by("-created_at")[:8]
    )

    recent_audit_logs = (
        AuditLog.objects.select_related("user")
        .order_by("-timestamp")[:10]
    )

    active_users = User.objects.filter(is_active=True).count()

    pending_count = Document.objects.filter(
        status__in=[
            Document.StatusChoices.SUBMITTED,
            Document.StatusChoices.UNDER_REVIEW,
        ]
    ).count()

    open_ticket_count = Ticket.objects.filter(
        status__in=[
            Ticket.StatusChoices.OPEN,
            Ticket.StatusChoices.IN_PROGRESS,
        ]
    ).count()

    for item in category_stats:
        item["label"] = dict(
            Asset.CategoryChoices.choices
        ).get(item["category"], item["category"])

        item["total_value"] = (
            item["total_value"] or Decimal("0.00")
        )

        if total_asset_value > 0:
            item["percentage"] = round(
                float(item["total_value"] / total_asset_value * 100),
                1,
            )
        else:
            item["percentage"] = 0

    context = {
        "total_system_assets": Asset.objects.count(),
        "active_asset_count": active_assets.count(),
        "total_asset_value": total_asset_value,
        "active_users": active_users,
        "inactive_users": User.objects.filter(
            is_active=False
        ).count(),
        "total_users": User.objects.count(),
        "total_clients": User.objects.filter(role="USER").count(),
        "admin_users": User.objects.filter(role="ADMIN").count(),
        "pending_count": pending_count,
        "open_ticket_count": open_ticket_count,
        "documents": pending_documents,
        "tickets": open_tickets,
        "category_stats": category_stats,
        "recent_assets": recent_assets,
        "recent_audit_logs": recent_audit_logs,
    }

    return render(
        request,
        "dashboards/admin_dashboard.html",
        context,
    )

@login_required
@admin_required
@require_http_methods(["GET", "POST"])
def admin_manage_users(request):
    """Search and update user access using validated POST requests."""

    if request.method == "POST":
        user_id = request.POST.get("user_id", "").strip()
        new_role = request.POST.get("role", "").strip()
        new_is_active = request.POST.get("is_active") == "on"

        valid_roles = {
            User.RoleChoices.USER,
            User.RoleChoices.ADMIN,
        }

        if not user_id.isdigit():
            messages.error(request, "Invalid user account.")
            return redirect("dashboards:admin_manage_users")

        if new_role not in valid_roles:
            messages.error(request, "The selected role is invalid.")
            return redirect("dashboards:admin_manage_users")

        with transaction.atomic():
            target_user = get_object_or_404(
                User.objects.select_for_update(),
                pk=int(user_id),
            )

            if target_user.pk == request.user.pk:
                messages.error(
                    request,
                    "You cannot change your own role or account status here.",
                )
                return redirect("dashboards:admin_manage_users")

            if target_user.is_superuser:
                messages.error(
                    request,
                    "Superuser accounts cannot be modified through this page.",
                )
                return redirect("dashboards:admin_manage_users")

            currently_active_admin = (
                target_user.is_active
                and target_user.role == User.RoleChoices.ADMIN
            )

            will_remain_active_admin = (
                new_is_active
                and new_role == User.RoleChoices.ADMIN
            )

            if currently_active_admin and not will_remain_active_admin:
                other_active_admins = User.objects.filter(
                    is_active=True,
                    role=User.RoleChoices.ADMIN,
                ).exclude(pk=target_user.pk).count()

                active_superusers = User.objects.filter(
                    is_active=True,
                    is_superuser=True,
                ).count()

                if other_active_admins + active_superusers == 0:
                    messages.error(
                        request,
                        "You cannot deactivate or demote the last active administrator.",
                    )
                    return redirect("dashboards:admin_manage_users")

            old_role = target_user.role
            old_is_active = target_user.is_active

            target_user.role = new_role
            target_user.is_active = new_is_active
            target_user.save(
                update_fields=["role", "is_active"]
            )

            if (
                old_role != new_role
                or old_is_active != new_is_active
            ):
                record_admin_action(
                    request,
                    "USER_ACCESS_UPDATED",
                    (
                        f"Target user ID: {target_user.pk}; "
                        f"role: {old_role} -> {new_role}; "
                        f"active: {old_is_active} -> {new_is_active}"
                    ),
                )

        messages.success(
            request,
            f"Access settings updated for {target_user.username}.",
        )

        return redirect("dashboards:admin_manage_users")

    search = request.GET.get("q", "").strip()

    users = User.objects.all().order_by("-date_joined")

    if search:
        from django.db.models import Q

        users = users.filter(
            Q(username__icontains=search)
            | Q(email__icontains=search)
            | Q(first_name__icontains=search)
            | Q(last_name__icontains=search)
        )
    context = {
        "users": users,
        "search": search,
        "total_users": users.count(),
        "active_users_count": User.objects.filter(
            is_active=True
        ).count(),
        "admin_users_count": User.objects.filter(
            role=User.RoleChoices.ADMIN
        ).count(),
    }

    return render(
        request,
        "dashboards/admin_manage_users.html",
        context,
    )
